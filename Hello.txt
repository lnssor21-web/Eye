local function ss(a, b, x)
	local t = (x - a) / (b - a)
	if t < 0 then t = 0 elseif t > 1 then t = 1 end
	return t * t * (3 - 2 * t)
end
local function ease(t)
	if t < 0 then t = 0 elseif t > 1 then t = 1 end
	return t * t * (3 - 2 * t)
end
local function mix(a, b, k) return a + (b - a) * k end

local function up2(base, g, n)
	local out = table.create(n * n, 0)
	for y = 0, n - 1 do
		local iv = y * g / n
		local j0 = floor(iv) % g
		local j1 = (j0 + 1) % g
		local fv = iv - floor(iv)
		fv = fv * fv * (3 - 2 * fv)
		local row = y * n
		local b0, b1 = j0 * g, j1 * g
		for x = 0, n - 1 do
			local iu = x * g / n
			local i0 = floor(iu) % g
			local i1 = (i0 + 1) % g
			local fu = iu - floor(iu)
			fu = fu * fu * (3 - 2 * fu)
			local t0 = base[b0 + i0 + 1] * (1 - fu) + base[b0 + i1 + 1] * fu
			local t1 = base[b1 + i0 + 1] * (1 - fu) + base[b1 + i1 + 1] * fu
			out[row + x + 1] = t0 * (1 - fv) + t1 * fv
		end
	end
	return out
end

local function normArr(arr, k)
	local n = #arr
	local s = 0
	for i = 1, n do s = s + arr[i] end
	local mean = s / n
	local v = 0
	for i = 1, n do local d = arr[i] - mean; v = v + d * d end
	local kk = sqrt(v / n) * (k or 4) + 1e-9
	for i = 1, n do arr[i] = clamp(0.5 + (arr[i] - mean) / kk, 0, 1) end
	return arr
end

local function tex2d(n, grids)
	local out = table.create(n * n, 0)
	local amp = 1
	for gi = 1, #grids do
		local g = grids[gi]
		local base = table.create(g * g)
		for i = 1, g * g do base[i] = rng:NextNumber() end
		local up = up2(base, g, n)
		for i = 1, n * n do out[i] = out[i] + amp * up[i] end
		amp = amp * 0.55
	end
	return normArr(out, 4)
end

local function noise1d(n, g)
	local base = table.create(g)
	for i = 1, g do base[i] = rng:NextNumber() end
	local out = table.create(n, 0)
	for i = 0, n - 1 do
		local x = i * g / n
		local i0 = floor(x) % g
		local i1 = (i0 + 1) % g
		local f = x - floor(x)
		f = f * f * (3 - 2 * f)
		out[i + 1] = base[i0 + 1] * (1 - f) + base[i1 + 1] * f
	end
	return out
end

local function bil(tex, u, v)
	local u0, v0 = floor(u), floor(v)
	local fu, fv = u - u0, v - v0
	local i0 = u0 % NT + 1
	local i1 = i0 % NT + 1
	local jb0 = (v0 % NT) * NT
	local jb1 = ((v0 % NT + 1) % NT) * NT
	local a = tex[jb0 + i0] * (1 - fu) + tex[jb0 + i1] * fu
	local b = tex[jb1 + i0] * (1 - fu) + tex[jb1 + i1] * fu
	return a * (1 - fv) + b * fv
end

local function blink_c(t, tb)
	tb = tb or T_BLINK
	local d = t - tb
	if d < 0 or d > 0.36 then return 0 end
	if d < 0.09 then return ease(d / 0.09) end
	if d < 0.12 then return 1 end
	return 1 - ease((d - 0.12) / 0.24)
end

local function lid_open(t)
	local k = ease((t - T_OPEN) / T_OPEN_D)
	k = k * (1 + 0.035 * sin(PI * k))
	return min(1, k) * (1 - 0.985 * blink_c(t))
end

local function gaze(t)
	local pgx, pgy, cgx, cgy, te = 0, 0, 0, 0, 0
	for i = 1, #GZ do
		local g = GZ[i]
		if g[1] <= t then
			pgx, pgy, cgx, cgy, te = cgx, cgy, g[2], g[3], g[1]
		else
			break
		end
	end
	local k = ease((t - te) / 0.09)
	local gx = pgx + (cgx - pgx) * k
	local gy = pgy + (cgy - pgy) * k
	gx = gx + 0.003 * sin(t * 1.3)
	gy = gy + 0.002 * sin(t * 1.1 + 1)
	local amp = 1 - ease((t - T_Z0) / 2.5)
	gy = gy - 0.03 * blink_c(t)
	return gx * amp, gy * amp
end

local function lidEdges(u0, ou)
	local s = u0 / WO
	local wu = s < -0.12 and 0.88 or 1.12
	local cu = max(0, 1 - ((s + 0.12) / wu) ^ 2) ^ 0.8
	local cl = max(0, 1 - (s - 0.08) ^ 2) ^ 0.9
	local vc = 0.03 * cl
	local vl = vc + (HL * cl - vc) * (0.35 + 0.65 * ou)
	local vu = vl + (-HU * cu - vl) * ou
	return vu, vl, cu, cl
end

local function heartNebula(xs, ys, sz, hm, t)
	local hx, hy = xs / sz, -ys / sz
	local nC = bil(TG, hx * 16 + 400, hy * 16 + 100)
	local nD = bil(TG, hx * 16 + 90, hy * 16 + 530)
	local nA = bil(TG, hx * 42 + 300, hy * 42 + 700 + t * 0.8)
	local nB = bil(TG, hx * 110 + 80, hy * 110 + 200)
	local wob = 1 - 0.65 * hm
	local hxw = hx + wob * 0.40 * (nC - 0.5) + 0.05 * (nA - 0.5)
	local hyw = hy + wob * 0.40 * (nD - 0.5) + 0.05 * (nA - 0.5)
	local qc = sqrt(hxw * hxw + hyw * hyw) / 0.80 - 1
	local hyh = hyw + 0.12
	local fh = (hxw * hxw + hyh * hyh - 1) ^ 3 - hxw * hxw * hyh ^ 3
	local qh = 0.5 * cbrt(fh)
	local q = qc * (1 - hm) + qh * hm
	local ins = clamp(-q / (0.80 - 0.35 * hm), 0, 1)
	local kk = 3 + 2.5 * hm
	local body
	if q < 0 then body = 0.50 + 0.50 * sqrt(ins) else body = exp(-clamp(q, 0, 10) * kk) * 0.50 end
	local tex = 1 + (0.62 + 0.80 * nB - 1) * 0.8
	local ik = ins ^ 0.8
	local bt = body * tex
	local cR, cG, cB = bt * mix(0.92, 1.0, ik), bt * mix(0.16, 0.58, ik), bt * mix(0.40, 0.78, ik)
	local halo = exp(-clamp(q, 0, 10) * 1.6) * 0.20 * (0.5 + nA) * (q > 0 and 1 or 0)
	cR = cR + halo * 0.78
	cG = cG + halo * 0.10
	cB = cB + halo * 0.30
	local rim = exp(-((q + 0.04) / 0.14) ^ 2)
	local rid = clamp((bil(TV, hxw * 120 + 50, hyw * 120 + 300 + t * 0.6) - 0.86) * 7, 0, 1)
	local rk = rim * (0.20 + 0.9 * rid) * (0.5 + nA)
	cR = cR + rk * 0.95
	cG = cG + rk * 0.16
	cB = cB + rk * 0.34
	local rk2 = rid * clamp(-q, 0, 1) * 0.25
	return cR + rk2 * 0.95, cG + rk2 * 0.16, cB + rk2 * 0.34
end

local function eyeScene(t, z, ou, gx, gy, Rp)
	local iz = 1 / z
	local px = iz / GH
	local near = z < 3.4

	for p = 1, NPIX do
		local u, v = Xs[p] * iz, Ys[p] * iz
		local pmU, pmV = u - gx, v - gy
		local r = sqrt(pmU * pmU + pmV * pmV)
		local duIn, dlIn, s, vt, vu, vl, mIn = 0, 0, 0, 0, 0, 0, 1
		if near then
			s = u / WO
			local sa = clamp(s, -1.3, 1.3)
			local wu = sa < -0.12 and 0.88 or 1.12
			local cu = max(0, 1 - ((sa + 0.12) / wu) ^ 2) ^ 0.8
			local cl = max(0, 1 - (sa - 0.08) ^ 2) ^ 0.9
			vt = v + 0.055 * s
			local vc = 0.03 * cl
			vl = vc + (HL * cl - vc) * (0.35 + 0.65 * ou)
			vu = vl + (-HU * cu - vl) * ou
			duIn = vt - vu
			dlIn = vl - vt
			mIn = clamp(min(duIn, dlIn) / (1.6 * px) + 0.5, 0, 1)
		end

		local ang = atan2(pmV, pmU)
		local ia0 = floor(((ang / TWO_PI) % 1) * NA) % NA
		local rn = clamp((r - Rp) / (RI - Rp), 0, 1.3)
		local rb = rn * 14
		local i0 = floor(rb)
		local fr = rb - i0
		local fib = 0.55 * A1[ia0 + 1] + 0.45 * (A2[(ia0 + i0 * 131) % NA + 1] * (1 - fr) + A2[(ia0 + (i0 + 1) * 131) % NA + 1] * fr)
		local zig = abs((((ang / TWO_PI) % 1) * 34) % 1 - 0.5) * 2 - 0.5
		local cr = 0.34 + 0.20 * (A3[ia0 + 1] - 0.5) + 0.035 * zig
		local mm = ss(-0.10, 0.22, rn - cr)
		local irR = mix(0.80, 0.42, mm)
		local irG = mix(0.45, 0.20, mm)
		local irB = mix(0.10, 0.06, mm)
		local f1 = 0.45 + 1.05 * fib
		irR = irR * f1
		irG = irG * f1
		irB = irB * f1
		local glow = exp(-((rn - cr) / 0.07) ^ 2) * (0.2 + fib)
		irR = irR + 0.22 * glow
		irG = irG + 0.11 * glow
		local f2 = 1 - 0.55 * ((A4[ia0 + 1] > 0.72) and ss(0.65, 0.18, rn) or 0)

		irB = irB * f3
		local f4 = 0.55 + 0.45 * ss(0, 0.07, rn)
		irR = irR * f4
		irG = irG * f4
		irB = irB * f4
		if near then
			local f5 = 0.30 + 0.70 * ss(0, 0.16, duIn)

			irG = irG * f5
			irB = irB * f5
		end

		if near then
			local cw = ss(0.40, 0.95, abs(s))

			scB = mix(scB, 0.14, mv)

			inR = mix(scR, irR, im)
			inG = mix(scG, irG, im)
			inB = mix(scB, irB, im)

			local crPos = 0.075 + 0.03 * (1 - ou)
			if upper then
				sh = sh * (1 - 0.42 * exp(-((du - crPos) / 0.020) ^ 2))
			else
				sh = sh * (1 + 0.08 * exp(-((dl - 0.09) / 0.04) ^ 2))
			end
			skR = skR * sh
			skG = skG * sh
			skB = skB * sh
			local lidzone = upper and ss(crPos, crPos * 0.4, du) or 0
			local ml = lidzone * 0.5
			skR = mix(skR, 0.68 * sh, ml)
			skG = mix(skG, 0.40 * sh, ml)
			skB = mix(skB, 0.31 * sh, ml)
			local sb = clamp(1 - abs(s), 0, 1)
			local eb = exp(-((vt + 0.42 + 0.10 * s * s) / (0.040 + 0.012 * sb)) ^ 2)
			local hair = 0.55 + 0.9 * bil(TW, u * 90, v * 30)
			local mh = clamp(eb * hair, 0, 0.92)
			skR = mix(skR, 0.10, mh)
			skG = mix(skG, 0.06, mh)
			skB = mix(skB, 0.04, mh)
			local line = exp(-(du / 0.010) ^ 2) * (upper and 1 or 0)
			local mln = clamp(line * (1 - 0.3 * ou) * 0.9, 0, 1)
			skR = mix(skR, 0.05, mln)
			skG = mix(skG, 0.03, mln)
			skB = mix(skB, 0.022, mln)
			local wl = (not upper) and ss(0.014, 0, dl) or 0
			local mw = wl * 0.7
			skR = mix(skR, 0.74, mw)
			skG = mix(skG, 0.34, mw)
			skB = mix(skB, 0.30, mw)
			local gl = exp(-((dl - 0.016) / 0.006) ^ 2) * ((not upper) and 1 or 0) * (0.35 + 0.65 * bil(TW, u * 60, v * 60)) * 0.55 * ss(0.12, 0.5, ou)
			skR = skR + gl
			skG = skG + gl * 0.85
			skB = skB + gl * 0.75
			local spk = ((bil(TW, u * 150 + 7, v * 150 + 3) > 0.985) and 0.35 or 0) * (0.4 + tx) * ss(0, 0.3, dd)
			skR = skR + spk
			skG = skG + spk * 0.9
			skB = skB + spk * 0.8
			inR = mix(skR, inR, mIn)
			inG = mix(skG, inG, mIn)
			inB = mix(skB, inB, mIn)
			local cg = exp(-(((u + WO * 0.90) / 0.045) ^ 2 + ((vt - 0.025) / 0.032) ^ 2)) * ss(0.1, 0.5, ou) * 0.85
			inR = mix(inR, 0.82, cg)
			inG = mix(inG, 0.42, cg)
			inB = mix(inB, 0.38, cg)
		end

		imgR[p], imgG[p], imgB[p] = inR, inG, inB
		pmArr[p] = pm
	end
end

local function updateWorld(t, dt, prog)
	local vf = 0.010 + 0.05 * prog + (t > T_SPACE and 0.015 or 0)
	for i = 1, NS do
		SD[i] = SD[i] - vf * dt
		if SD[i] < 0.06 then
			SD[i] = SD[i] + 0.94
			SX[i] = rng:NextNumber(-1, 1)
			SY[i] = rng:NextNumber(-1, 1)
		end
	end
	if S.nextShoot > 1e8 and t > T_SPACE - 4 then S.nextShoot = t + 1 end
	if t > S.nextShoot then
		S.nextShoot = t + prnd:NextNumber(2.2, 4.5)
		local ang = rad(prnd:NextNumber(18, 62))
		local sgn = prnd:NextInteger(0, 1) == 0 and -1 or 1
		local spd = prnd:NextNumber(0.7, 1.2)
		table.insert(streaks, {
			x = prnd:NextNumber(-0.7, 0.7), y = prnd:NextNumber(-0.45, -0.15),
			vx = sgn * cos(ang) * spd, vy = sin(ang) * spd,
			life = prnd:NextNumber(0.9, 1.4), tail = prnd:NextNumber(0.14, 0.26),
			colR = 0.88, colG = 0.94, colB = 1.0, soft = false, age = 0,
		})
	end
	if t > T_SPACE + 1.5 and not S.comet then
		S.comet = true
		table.insert(streaks, { x = -0.95, y = -0.28, vx = 0.030, vy = 0.008, life = 70, tail = 0.22, colR = 0.70, colG = 0.86, colB = 1.0, soft = true, age = 0 })
	end
	local n = 0
	for i = 1, #streaks do
		local st = streaks[i]
		st.age = st.age + dt
		if st.age < st.life then
			n = n + 1
			streaks[n] = st
		end
	end
	for i = #streaks, n + 1, -1 do streaks[i] = nil end
end

local function putCell(c, chIdx, cr, cg, cb, a)
	idx[c] = chIdx
	fgR[c] = cr * a
	fgG[c] = cg * a
	fgB[c] = cb * a
end

local function stampStreak(st)
	local hx = st.x + st.vx * st.age
	local hy = st.y + st.vy * st.age
	local spd = sqrt(st.vx * st.vx + st.vy * st.vy)
	local dx, dy = st.vx / spd, st.vy / spd
	local fade = sin(PI * min(1, st.age / st.life)) ^ 0.7
	if fade < 0.02 then return end
	local tc
	if abs(dy) < 0.38 * abs(dx) then tc = "-"
	elseif abs(dx) < 0.38 * abs(dy) then tc = "|"
	elseif dx * dy > 0 then tc = "\\"
	else tc = "/" end
	local n = max(6, floor(st.tail * H / (CW * 0.8)))
	for k = 0, n - 1 do
		local f = k / n
		local c = floor((W / 2 + (hx - dx * st.tail * f) * H) / CW)
		local r = floor((H / 2 + (hy - dy * st.tail * f) * H) / CH)
		local b = (1 - f) ^ 1.3 * fade
		if b >= 0.08 and r >= 0 and r < ROWS and c >= 0 and c < COLS then
			local ci = r * COLS + c + 1
			if pmc[ci] >= 0.5 then
				local chIdx
				if k == 0 then chIdx = CIDX[st.soft and "*" or "@"]
				elseif st.soft then chIdx = CIDX[b > 0.35 and ":" or "."]
				else chIdx = CIDX[b > 0.30 and tc or "."] end
				putCell(ci, chIdx, st.colR, st.colG, st.colB, min(1, 0.3 + b) * vigc[ci])
			end
		end
	end
end

local function stampStars(t, gs, amt, offx, offy)
	if amt <= 0.003 then return end
	for i = 1, NS do
		local d = SD[i]
		local c = floor((W * 0.5 + (SX[i] * gs / d + offx) * H) / CW)
		local r = floor((H * 0.5 + (SY[i] * gs / d + offy) * H) / CH)
		if c >= 0 and c < COLS and r >= 0 and r < ROWS then
			local alpha = ss(1.0, 0.82, d) * (0.80 + 0.20 * sin(t * SF[i] + SPH[i])) * amt
			if alpha > 0.04 then
				local ci = r * COLS + c + 1
				if pmc[ci] > 0.5 then
					local b = SB[i] * alpha * (1 + 0.05 / d)
					if b > cellLi[ci] * 0.7 then
						local chIdx
						if b < 0.20 then chIdx = CIDX["."]
						elseif b < 0.42 then chIdx = CIDX["+"]
						else chIdx = CIDX["*"] end
						idx[ci] = chIdx
						local fk = clamp(0.35 + 0.9 * b, 0, 1) * vigc[ci]
						fgR[ci] = SCOLr[i] * fk
						fgG[ci] = SCOLg[i] * fk
						fgB[ci] = SCOLb[i] * fk
						if b > 0.55 then
							idx[ci] = b > 0.85 and CIDX["@"] or CIDX["*"]
							local ln = floor(2 + 6 * (b - 0.5))
							local vr = 1
							for k = 1, ln do
								local a = (1 - k / (ln + 1)) * b
								vr = k // 2 + (k > 2 and 1 or 0)
								for ai = 1, 4 do
									local dr, dc, chh
									if ai == 1 then dr, dc, chh = 0, k, "-"
									elseif ai == 2 then dr, dc, chh = 0, -k, "-"
									elseif ai == 3 then dr, dc, chh = vr, 0, "|"
									else dr, dc, chh = -vr, 0, "|" end
									local r2, c2 = r + dr, c + dc
									if r2 >= 0 and r2 < ROWS and c2 >= 0 and c2 < COLS then
										local ci2 = r2 * COLS + c2 + 1
										if pmc[ci2] > 0.5 and cellLi[ci2] < a * 0.8 then
											idx[ci2] = CIDX[a > 0.18 and chh or "."]
											local fk2 = vigc[ci2] * min(1, 0.25 + a)
											fgR[ci2] = SCOLr[i] * fk2
											fgG[ci2] = SCOLg[i] * fk2
											fgB[ci2] = SCOLb[i] * fk2
										end
									end
								end
							end
						end
					end
				end
			end
		end
	end
end

local function stampLashes(z, ou)
	if z > 2.6 or ou < 0 then return end
	local cl = 1 - ou ^ 0.8
	local function pick(aa)
		local ang = deg(atan2(-sin(aa), cos(aa))) % 180
		if ang < 25 or ang >= 155 then return "-" end
		if ang < 70 then return "/" end
		if ang < 110 then return "|" end
		return "\\"
	end
	local function draw(baseU, baseV, a0, L, curl, crR, crG, crB, ctR, ctG, ctB, nmax, tip)
		local n = floor(max(2, min(nmax, L * z * H / (CW * 0.85))))
		local step = L / n
		local ue, ve = baseU, baseV
		for k = 0, n - 1 do
			local tt = k / max(1, n - 1)
			local aa = a0 + curl * tt * tt
			ue = ue + cos(aa) * step
			ve = ve + sin(aa) * step
			local c = floor((W / 2 + ue * z * H) / CW)
			local r = floor((H / 2 + ve * z * H) / CH)
			if r >= 0 and r < ROWS and c >= 0 and c < COLS then
				local ci = r * COLS + c + 1
				idx[ci] = (k == n - 1 and n >= 4) and CIDX[tip] or CIDX[pick(aa)]
				fgR[ci] = crR + (ctR - crR) * tt
				fgG[ci] = crG + (ctG - crG) * tt
				fgB[ci] = crB + (ctB - crB) * tt
			end
		end
	end
	for i = 1, NUP do
		local e = UP_L[i]
		local s0, lf = e[1], e[2]
		local vu = lidEdges(s0 * WO, ou)
		local aOpen = rad(-90 + 26 * s0)
		local aClosed = s0 >= 0 and rad(90 - 20 * s0) or rad(-270 - 20 * s0)
		local a0 = aOpen + (aClosed - aOpen) * cl
		local edge = sqrt(max(0, 1 - s0 * s0))
		draw(s0 * WO, vu - 0.055 * s0, a0, (0.070 + 0.050 * edge) * lf, rad(-22 * s0) * (1 - cl), 0.16, 0.10, 0.07, 0.30, 0.19, 0.13, 14, "'")
	end
	for i = 1, NLO do
		local e = LO_L[i]
		local s0, lf = e[1], e[2]
		local vu, vl, cu, cll = lidEdges(s0 * WO, ou)
		draw(s0 * WO, vl - 0.055 * s0, rad(90 - 20 * s0), 0.034 * (0.5 + 0.8 * cll) * lf, rad(12 * (s0 > 0 and 1 or -1)), 0.17, 0.11, 0.08, 0.30, 0.20, 0.14, 6, ".")
	end
end

local function bloom()
	local function lvl(f, wgt)
		local w2, h2 = floor(GW / f), floor(GH / f)
		if w2 < 2 or h2 < 2 then return end
		local inv = 1 / (f * f)
		for y = 0, h2 - 1 do
			for x = 0, w2 - 1 do
				local sr, sg, sb = 0, 0, 0
				for oy = 0, f - 1 do
					local base = (y * f + oy) * GW
					for ox = 0, f - 1 do
						local p = base + x * f + ox + 1
						sr = sr + bR[p]
						sg = sg + bG[p]
						sb = sb + bB[p]
					end
				end
				local o = y * w2 + x + 1
				lvR[o], lvG[o], lvB[o] = sr * inv, sg * inv, sb * inv
			end
		end
		for _ = 1, 2 do
			for y = 0, h2 - 1 do
				local up = ((y - 1) % h2) * w2
				local dn = ((y + 1) % h2) * w2
				local cur = y * w2
				for x = 0, w2 - 1 do
					local xl = (x - 1) % w2
					local xr = (x + 1) % w2
					local o = cur + x + 1
					tpR[o] = (lvR[o] * 2 + lvR[up + x + 1] + lvR[dn + x + 1] + lvR[cur + xl + 1] + lvR[cur + xr + 1]) / 6
					tpG[o] = (lvG[o] * 2 + lvG[up + x + 1] + lvG[dn + x + 1] + lvG[cur + xl + 1] + lvG[cur + xr + 1]) / 6
					tpB[o] = (lvB[o] * 2 + lvB[up + x + 1] + lvB[dn + x + 1] + lvB[cur + xl + 1] + lvB[cur + xr + 1]) / 6
				end
			end
			for i = 1, w2 * h2 do lvR[i], lvG[i], lvB[i] = tpR[i], tpG[i], tpB[i] end
		end
		for y = 0, GH - 1 do
			local sy = (y // f) * w2
			local row = y * GW
			for x = 0, GW - 1 do
				local p = row + x + 1
				local s = sy + (x // f) + 1
				imgR[p] = imgR[p] + wgt * lvR[s]
				imgG[p] = imgG[p] + wgt * lvG[s]
				imgB[p] = imgB[p] + wgt * lvB[s]
			end
		end
	end
	for p = 1, NPIX do
		bR[p] = max(imgR[p] - 0.45, 0)
		bG[p] = max(imgG[p] - 0.45, 0)
		bB[p] = max(imgB[p] - 0.45, 0)
	end
	lvl(2, 0.40)
	lvl(4, 0.30)
end

local function renderCells(t, dt)
	local prog = ease((t - T_Z0) / T_ZD)
	local idle = ease(t / T_Z0)
	local z = exp(math.log(ZEND) * prog) * (1 + 0.05 * idle + 0.004 * sin(t * 0.9))
	local zsp = t < T_SPACE and z or ZEND * exp(0.03 * (t - T_SPACE))
	local gs = min(6, GS0 * zsp ^ 1.25)
	local gh = min(1.15, GS0 * z ^ 1.25)
	local ou = lid_open(t)
	local gx, gy = gaze(t)
	local Rp = RP0 * (0.76 + 0.24 * ease((t - 1.8) / 3.5)) * (1 + 0.025 * sin(1.9 * t)) * (1 + 0.30 * prog)
	local Rps = Rp * z
	local full = Rps * 0.72 > 1.03
	local starAmt = ease((t - (T_Z0 + 1.5)) / 4.5)
	local nebAmt = ease((t - (T_Z0 + 3.5)) / 8.0)
	updateWorld(t, dt, prog)
	local offx, offy = gx * z, gy * z

	for p = 1, NPIX do
		spR[p] = 0.004
		spG[p] = 0.005
		spB[p] = 0.012
	end
	if nebAmt > 0.002 then
		local drift = ease((t - T_SPACE) / 20)
		local hm = ease((t - T_HEART) / T_HEART_D)
		local beat = 1 + 0.03 * sin(t * 2.6) * hm * hm
		local kk = 1.25 * nebAmt ^ 1.3
		local sz = 0.17 * gh * beat
		local wob = sin(t * 0.07) * 0.03 * drift
		for p = 1, NPIX do
			local hr, hg, hb = heartNebula(Xs[p] - offx - wob, Ys[p] - offy, sz, hm, t)
			spR[p] = spR[p] + hr * kk
			spG[p] = spG[p] + hg * kk
			spB[p] = spB[p] + hb * kk
		end
	end

	if full then
		for p = 1, NPIX do
			imgR[p], imgG[p], imgB[p] = spR[p], spG[p], spB[p]
			pmArr[p] = 1
		end
		for i = 1, NCELLS do
			pmc[i] = 1
			solid[i] = false
		end
	else
		eyeScene(t, z, ou, gx, gy, Rp)
		local inv = 1 / (SS * SS)
		for r = 0, ROWS - 1 do
			for c = 0, COLS - 1 do
				local i = r * COLS + c + 1
				local acc2 = 0
				for oy = 0, SS - 1 do
					local base = (r * SS + oy) * GW + c * SS
					for ox = 0, SS - 1 do
						acc2 = acc2 + pmArr[base + ox + 1]
					end
				end
				local v = acc2 * inv
				pmc[i] = v
				solid[i] = v < 0.5
			end
		end
	end

	bloom()

	local fadeIn = ease(t / 1.2)
	for p = 1, NPIX do
		local v = (1 - 0.30 * ss(0.4, 1.0, RSs[p])) * fadeIn
		imgR[p] = (1 - exp(-2 * max(imgR[p], 0))) * v
		imgG[p] = (1 - exp(-2 * max(imgG[p], 0))) * v
		imgB[p] = (1 - exp(-2 * max(imgB[p], 0))) * v
	end

	local inv = 1 / (SS * SS)
	for r = 0, ROWS - 1 do
		for c = 0, COLS - 1 do
			local i = r * COLS + c + 1
			local cr, cg, cb = 0, 0, 0
			for oy = 0, SS - 1 do
				local base = (r * SS + oy) * GW + c * SS
				for ox = 0, SS - 1 do
					local p = base + ox + 1
					cr = cr + imgR[p]
					cg = cg + imgG[p]
					cb = cb + imgB[p]
				end
			end
			cr = cr * inv
			cg = cg * inv
			cb = cb * inv
			local lum = 0.30 * cr + 0.60 * cg + 0.10 * cb
			local li = clamp(lum, 0, 1) ^ 0.72
			cellLi[i] = li
			local ri = floor(clamp((li + (HSC[i] - 0.5) * 0.07) * NR, 0, NR))
			if solid[i] and ri < 3 then ri = 3 end
			idx[i] = ri
			local fr, fg, fb = clamp(cr * 1.15, 0, 1), clamp(cg * 1.15, 0, 1), clamp(cb * 1.15, 0, 1)
			fgR[i] = fr
			fgG[i] = fg
			fgB[i] = fb
			qR[i] = clamp(floor(fr * (QL - 0.01)), 0, QL - 1)
			qG[i] = clamp(floor(fg * (QL - 0.01)), 0, QL - 1)
			qB[i] = clamp(floor(fb * (QL - 0.01)), 0, QL - 1)
			vigc[i] = 0.25 + 0.75 * ss(0, 0.30 * Rps, Rps - RSC[i])
		end
	end

	stampStars(t, gs, starAmt, offx, offy)
	for i = 1, #streaks do stampStreak(streaks[i]) end
	if not full then stampLashes(z, ou) end
end

local function playSound()
	local fileName = "Swp29s.mp3"
	local soundName = "Swp29s_Local"
	local lockKey = "_DeltaSwp29sPlaying"

	if SoundService:GetAttribute(lockKey) then return end
	SoundService:SetAttribute(lockKey, true)
	task.delay(0.5, function()
		SoundService:SetAttribute(lockKey, false)
	end)

	for _, obj in ipairs(SoundService:GetDescendants()) do
		if obj:IsA("Sound") and obj.Name == soundName then
			pcall(function()
				obj:Stop()
				obj:Destroy()
			end)
		end
	end

	if type(getcustomasset) ~= "function" then
		warn("[Swp29s] getcustomasset not available.")
		return
	end

	local ok, asset = pcall(getcustomasset, fileName)
	if not ok or not asset or asset == "" then
		warn("[Swp29s] Failed to load asset:", tostring(asset))
		return
	end

	local s = Instance.new("Sound")
	s.Name = soundName
	s.Volume = CONFIG.SOUND_VOLUME
	s.PlaybackSpeed = 1
	s.TimePosition = 0
	s.Looped = false
	s.SoundId = asset
	s.Parent = SoundService

	sound = s

	if not s.IsLoaded then
		s.Loaded:Wait()
	end
	if not running then return end

	s:Play()

	s.Ended:Connect(function()
		s:Destroy()
	end)
end

local function close()
	if not running then return end
	running = false
	if conn then conn:Disconnect() end
	if spawnThread then pcall(task.cancel, spawnThread) end
	if sound then
		pcall(function()
			sound:Stop()
			sound:Destroy()
		end)
		sound = nil
	end
	gui:Destroy()
	table.clear(streaks)
	table.clear(colorCache)
end

local t0 = os.clock()
local acc = 0
local step = 1 / CONFIG.FPS
local lastT = 0
local budget = 1500

conn = RunService.RenderStepped:Connect(function(dt)
	if not running then return end
	acc = acc + dt
	if acc < step then return end
	acc = 0
	local frameStart = os.clock()
	local t = (frameStart - t0) * CONFIG.SPEED
	local d = clamp(t - lastT, 0, 0.1)
	lastT = t

	renderCells(t, d)

	local list = {}
	local n = 0
	for i = 1, NCELLS do
		local ni, di = idx[i], dIdx[i]
		local same = ni == di
		local dq = max(abs(qR[i] - dQR[i]), max(abs(qG[i] - dQG[i]), abs(qB[i] - dQB[i])))
		if not (ni == 0 and di == 0) and (not same or dq >= 2) then
			n = n + 1
			list[n] = i
			prio[i] = (same and 0 or 6) + dq
		end
	end
	if n > budget then
		table.sort(list, byPrio)
		for i = n, budget + 1, -1 do list[i] = nil end
		n = budget
	end
	for i = 1, n do
		local c = list[i]
		local ni = idx[c]
		labels[c].Text = RAMP:sub(ni + 1, ni + 1)
		labels[c].TextColor3 = colorFor(qR[c], qG[c], qB[c])
		dIdx[c] = ni
		dQR[c], dQG[c], dQB[c] = qR[c], qG[c], qB[c]
	end

	if n > 40 then
		local el = os.clock() - frameStart
		budget = clamp(floor(0.75 * budget + 0.25 * (0.020 / max(el, 1e-4))), 300, 8000)
	end
end)
